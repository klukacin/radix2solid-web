## Anatomy

Import all parts and piece them together.

```jsx
import { Dialog } from "@wsagency/radix2solid-primitives";

export default () => (
	<Dialog.Root>
		<Dialog.Trigger />
		<Dialog.Portal>
			<Dialog.Overlay />
			<Dialog.Content>
				<Dialog.Title />
				<Dialog.Description />
				<Dialog.Close />
			</Dialog.Content>
		</Dialog.Portal>
	</Dialog.Root>
);
```
For a custom trigger, `asChild` takes a target factory, not an already-created JSX child.
The target attaches the primitive's props, ref, and event composition to the selected host.

```tsx
<Dialog.Trigger asChild>
	{target => target("button", { class: "custom-trigger", children: "Open dialog" })}
</Dialog.Trigger>
```

### Content

Contains content to be rendered in the open dialog.

`Dialog.Content` sets `aria-labelledby` to `Dialog.Title` and `aria-describedby` to `Dialog.Description` using IDs that `Dialog.Root` allocates up front, so it never inspects its children. Passing `aria-labelledby` or `aria-describedby` to Content replaces that default. Passing either one as `undefined` removes that relationship. A nonblank `aria-label` names Content instead of the title, while an `undefined` or empty `aria-label` keeps the title. A Title or Description with its own `id` isn't linked automatically: pass the same ID to Content, e.g. `<Dialog.Content aria-labelledby="profile-title">` with `<Dialog.Title id="profile-title">`.

**Breaking in 0.1.0-rc.2:** Content no longer collects part IDs or appends them to the relationships you pass. Add Content ARIA for custom part IDs, list the title ID yourself when combining it with an external label, and pass `aria-describedby={undefined}` when there is no Description.

### Title

An accessible title to be announced when the dialog is opened.

If you want to hide the title, use our [Visually Hidden](../utilities/visually-hidden) utility: `<VisuallyHidden.Root asChild>{target => target(Dialog.Title, { children: "Title" })}</VisuallyHidden.Root>`. A custom `id` replaces the Root ID, so link it with `aria-labelledby` on `Dialog.Content`.

## Examples

### Close after asynchronous form submission

Use the controlled props to programmatically close the Dialog after an async operation has completed.

```jsx line=4,7,10,15,17
import { createSignal } from "solid-js";
import { Dialog } from "@wsagency/radix2solid-primitives";

const wait = () => new Promise((resolve) => setTimeout(resolve, 1000));

export default () => {
	const [open, setOpen] = createSignal(false);

	return (
		<Dialog.Root __open__={open()} __onOpenChange__={setOpen}>
			<Dialog.Trigger>Open</Dialog.Trigger>
			<Dialog.Portal>
				<Dialog.Overlay />
				<Dialog.Content aria-label="Submit form" aria-describedby={undefined}>
					<form
						onSubmit={(event) => {
							wait().then(() => setOpen(false));
							event.preventDefault();
						}}
					>
						{/** some inputs */}
						<button type="submit">Submit</button>
					</form>
				</Dialog.Content>
			</Dialog.Portal>
		</Dialog.Root>
	);
};
```

### Scrollable overlay

Move the content inside the overlay to render a dialog with overflow.

```jsx
// index.tsx
import { Dialog } from "@wsagency/radix2solid-primitives";
import "./styles.css";

export default () => {
	return (
		<Dialog.Root>
			<Dialog.Trigger />
			<Dialog.Portal>
				<Dialog.Overlay class="DialogOverlay">
					<Dialog.Content class="DialogContent">...</Dialog.Content>
				</Dialog.Overlay>
			</Dialog.Portal>
		</Dialog.Root>
	);
};
```

### Custom portal container

Customise the element that your dialog portals into.

```tsx line=5,10,16
import { createSignal } from "solid-js";
import { Dialog } from "@wsagency/radix2solid-primitives";

export default () => {
	const [container, setContainer] = createSignal<HTMLDivElement | null>(null);
	return (
		<div>
			<Dialog.Root>
				<Dialog.Trigger />
				<Dialog.Portal __container__={container()}>
					<Dialog.Overlay />
					<Dialog.Content>...</Dialog.Content>
				</Dialog.Portal>
			</Dialog.Root>

			<div ref={node => { __setContainer__(node); }} />
		</div>
	);
};
```

## Custom APIs

Create your own API by abstracting the primitive parts into your own component.

### Abstract the overlay and the close button

This example abstracts the `Dialog.Overlay` and `Dialog.Close` parts.

#### Usage

```jsx
import { Dialog, DialogTrigger, DialogContent } from "./your-dialog";

export default () => (
	<Dialog>
		<DialogTrigger>Dialog trigger</DialogTrigger>
		<DialogContent aria-label="Dialog" aria-describedby={undefined}>Dialog Content</DialogContent>
	</Dialog>
);
```

#### Implementation

```tsx
// your-dialog.tsx
import { omit } from "solid-js";
import type { ComponentProps } from "@solidjs/web";
import { Dialog as DialogPrimitive } from "@wsagency/radix2solid-primitives";
import { Cross1Icon } from "@wsagency/radix2solid-icons";

export function DialogContent(props: Extract<ComponentProps<typeof DialogPrimitive.Content>, { asChild?: false }>) {
	return (
		<DialogPrimitive.Portal>
			<DialogPrimitive.Overlay />
			<DialogPrimitive.Content {...omit(props, "children")}>
				{props.children}
				<DialogPrimitive.Close aria-label="Close">
					<Cross1Icon />
				</DialogPrimitive.Close>
			</DialogPrimitive.Content>
		</DialogPrimitive.Portal>
	);
}

export const Dialog = DialogPrimitive.Root;
export const DialogTrigger = DialogPrimitive.Trigger;
```
