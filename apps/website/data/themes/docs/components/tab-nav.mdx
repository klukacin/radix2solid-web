## Examples

Use the `size` prop to control the size of the tabs.


### With router links

Use the `asChild` prop to compose `TabNav.Link` with your app’s router link component.

```tsx
export default (props: { pathname: string }) => <TabNav.Root>
	<TabNav.Link asChild active={props.pathname === "/account"}>{target => target("a", {
        "href": "/account",
        children: <>Account</>
      })}</TabNav.Link>
	<TabNav.Link asChild active={props.pathname === "/documents"}>{target => target("a", {
        "href": "/documents",
        children: <>Documents</>
      })}</TabNav.Link>
	<TabNav.Link asChild active={props.pathname === "/settings"}>{target => target("a", {
        "href": "/settings",
        children: <>Settings</>
      })}</TabNav.Link>
</TabNav.Root>;
```
